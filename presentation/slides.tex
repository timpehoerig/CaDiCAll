\begin{frame}{\textcolor{green}{\textbf{Model Enumeration}}}
  \center{cnf: $(X \lor Y) \land (\bar{Y} \lor Z)$}
  \begin{align*}
    solution \ 1: \ &X \land Y \land Z \\
    solution \ 2: \ &\bar{X} \land Y \land Z \\
    solution \ 3: \ &X \land \bar{Y} \land Z \\
    solution \ 4: \ &X \land \bar{Y} \land \bar{Z} \\
  \end{align*}
\end{frame}

\begin{frame}{\textcolor{green}{\textbf{Model Enumeration}} using \textcolor{orange}{\textbf{Blocking Clauses}}}
  \textbf{Pseudo Code}
  \begin{beamerbox}[Beispielbox]
    \begin{tabular}{l}
    \texttt{\textcolor{cyan}{list}[\textcolor{cyan}{model}] ms} \\
    \texttt{\textcolor{magenta}{while} solver.\textcolor{orange}{solve}(cnf) == SAT:}\\
    \texttt{\quad \textcolor{cyan}{model} m = solver.\textcolor{orange}{solution}()}\\
    \texttt{\quad ms += [m]}\\
    \texttt{\quad cnf += \textcolor{orange}{not}(m)}
    \end{tabular}
  \end{beamerbox}
\end{frame}

\begin{frame}{\textcolor{green}{\textbf{Model Enumeration}} using \textcolor{orange}{\textbf{Blocking Clauses}}}
  \textbf{Blocking Clause}
  \begin{align*}
  cnf&: (X \lor Y) \land (\bar{Y} \lor Z)\\
  model&: X \land Y \land Z\\
  \textcolor{red}{not}(model)&: \neg (X \land Y \land Z) \iff \bar{X} \lor \bar{Y} \lor \bar{Z}\\
  new \ cnf&: (X \lor Y) \land (\bar{Y} \lor Z) \land \textcolor{orange}{(\bar{X} \lor \bar{Y} \lor \bar{Z})}
  \end{align*}
\end{frame}

\begin{frame}{\textcolor{green}{\textbf{Model Enumeration}} using \textcolor{red}{\textbf{IPASIR-UP}}}
  \textbf{IPASIR-UP} extends IPASIR by an User Propagator
  \begin{itemize}
  \item cb\_check\_found\_model
  \item cb\_has\_external\_clause
  \item cb\_add\_external\_clause\_lit
  \item notify\_assignment
  \item notify\_new\_decision\_level
  \item notify\_backtrack
  \item cb\_decide
  \item cb\_propagate
  \item cb\_add\_reason\_clause\_lit
  \end{itemize}
\end{frame}

\begin{frame}{\textcolor{green}{\textbf{Model Enumeration}} \textbf{without} \textcolor{orange}{\textbf{Blocking Clauses}}}

\begin{frame}{\textcolor{pink}{\textbf{Implicant shrinking}}}
  \textbf{2-watching literal scheme}

  \[
  \begin{aligned}
    \\
    \texttt{cnf: }&(\underset{\uparrow}{X} \lor \underset{\uparrow}{Y}) \land (\underset{\uparrow}{\bar{Y}} \lor \underset{\uparrow}{Z})\\
    \\
    \texttt{cnf': }&(\underset{\uparrow}{X} \lor \underset{\uparrow}{Y}) \land (\underset{\uparrow}{\bar{Y}} \lor \underset{\uparrow}{Z} \lor X)
  \end{aligned}
  \]
\end{frame}

\begin{frame}{\textcolor{pink}{\textbf{Implicant shrinking}}}
  \textbf{Access to watches}
  \begin{beamerbox}[Beispielbox]
    \begin{tabular}{l}
    \texttt{\textcolor{cyan}{int} \textcolor{orange}{check\_literal}(\textcolor{cyan}{int} v, \textcolor{cyan}{int} b, ...): } \\
    \texttt{\quad \textcolor{magenta}{for} lit \textcolor{magenta}{in} \textcolor{pink}{other\_watching\_literal(v)}:} \\
    \texttt{\quad \quad \textcolor{magenta}{if} !sat(lit, \textcolor{orange}{get\_n}(stack, index)):} \\
    \texttt{\quad \quad \quad b = max(b, decision\_level(l))} \\
    \texttt{\quad \textcolor{magenta}{return} b}
    \end{tabular}
  \end{beamerbox}
\end{frame}

\begin{frame}{Drawbacks}
  \begin{itemize}
    \item Maintaining chronological order (ask\_decision)
    \item Out of order propagations
    \item No access to watches
    \item Optimizations: Interacting with the solver
    \item Optimizations: Duplication of data
  \end{itemize}
\end{frame}
