\begin{frame}{\textcolor{green}{\textbf{Model Enumeration}} using \textcolor{orange}{\textbf{Blocking Clauses}}}
  \textbf{Blocking Clause}
  \begin{align*}
  cnf&: (X \lor Y) \land (\bar{Y} \lor Z)\\
  model&: X \land Y \land Z\\
  \textcolor{red}{not}(model)&: \neg (X \land Y \land Z) \iff \bar{X} \lor \bar{Y} \lor \bar{Z}\\
  new \ cnf&: (X \lor Y) \land (\bar{Y} \lor Z) \land \textcolor{orange}{(\bar{X} \lor \bar{Y} \lor \bar{Z})}
  \end{align*}
\end{frame}

\begin{frame}{\textbf{\textcolor{pink}{Proof} about my program}}
  \begin{itemize}
    \item no identical models
    \item if minimized: models differ
    \item if not minimized: models are total
    \item if projected: cnf with model assumption is SAT
    \item if not projected: models satisfy cnf
  \end{itemize}
\end{frame}

\begin{frame}{\textbf{\textcolor{pink}{Proof} no identical models}}
  Loop over all models. \\
  Add every model to a set, if it is not already in there. \\
  Also check that no literal occurs twice.
\end{frame}

\begin{frame}{\textbf{\textcolor{pink}{Proof} if minimized: models differ}}
  Check that all models pairwise contain at least one literal that occurs negative in the other.
\end{frame}

\begin{frame}{\textbf{\textcolor{pink}{Proof} if not minimized: models are total}}
  Check that all models contain all (projected) literals.
\end{frame}

\begin{frame}{\textbf{\textcolor{pink}{Proof} if projected: cnf with model assumption is SAT}}
  For each model:\\
  \quad Create a new solver instance. \\
  \quad Assume all literals from one model. \\
  \quad Check that it is still SAT.
\end{frame}

\begin{frame}{\textbf{\textcolor{pink}{Proof} if not projected: models satisfy cnf}}
  For each model:\\
  \quad Check that it satisfies the cnf.
\end{frame}
